import React from 'react';
import { BarChart, Bar, XAxis, YAxis, CartesianGrid, Tooltip, Legend, ResponsiveContainer } from 'recharts';

const defaultChartData = [
  { name: 'Category A', value: 100 },
  { name: 'Category B', value: 200 },
  { name: 'Category C', value: 150 },
  { name: 'Category D', value: 300 },
];

interface BarChart3Props {
  width?: number;
  height?: number;
  data: any[]; // Expects an array of objects like { name: string, value: number }
}

const BarChart3: React.FC<BarChart3Props> = ({ width = 500, height = 300, data }) => {
  // Use provided data or default data if empty/null
  const chartData = data && data.length > 0 ? data : defaultChartData;

  return (
    <ResponsiveContainer width="100%" height="100%">
      <BarChart
        data={chartData}
        margin={{
          top: 5,
          right: 30,
          left: 20,
          bottom: 5,
        }}
      >
        <CartesianGrid strokeDasharray="3 3" />
        <XAxis dataKey="name" />
        <YAxis />
        <Tooltip />
        <Legend />
        <Bar dataKey="value" fill="#8884d8" name="Value" />
      </BarChart>
    </ResponsiveContainer>
  );
};

export default BarChart3;
